    function isVisibleElement(element) {
        if (!element) {
            return false;
        }

        var styles =
            window.getComputedStyle(element);

        var rect =
            element.getBoundingClientRect();

        return (
            styles.display !== "none" &&
            styles.visibility !== "hidden" &&
            Number(styles.opacity || "1") !== 0 &&
            rect.width > 0 &&
            rect.height > 0
        );
    }

    function normaliseUkMobileNumber(value) {
        return String(value || "")
            .trim()
            .replace(/[\s().-]/g, "");
    }

    function isValidUkMobileNumber(value) {
        var raw =
            String(value || "").trim();

        if (
            !raw ||
            !/^[0-9+\s().-]+$/.test(raw)
        ) {
            return false;
        }

        var normalised =
            normaliseUkMobileNumber(raw);

        return (
            /^07\d{9}$/.test(normalised) ||
            /^\+447\d{9}$/.test(normalised) ||
            /^00447\d{9}$/.test(normalised)
        );
    }

    function normaliseValidationMessage(value) {
        return String(value || "")
            .replace(/\s+/g, " ")
            .trim()
            .toLowerCase();
    }

    function isNativeMobileNumberError(
        element
    ) {
        if (
            !element ||
            element.tagName === "INPUT"
        ) {
            return false;
        }

        var text =
            normaliseValidationMessage(
                element.textContent
            );

        return (
            text.indexOf(
                NATIVE_MOBILE_ERROR_TEXT
            ) !== -1 &&
            text.length <= 80
        );
    }

    function findNativeMobileNumberError(
        field
    ) {
        if (!field) {
            return null;
        }

        var localRoot =
            field.closest(
                ".form-group, " +
                ".form-row, " +
                ".field, " +
                ".registrationField, " +
                ".registrationRow, " +
                ".inputArea, " +
                ".control-group, " +
                "li, td"
            ) ||
            field.parentElement;

        var selectors =
            "span, label, p, small, em, strong, div";

        var candidates;
        var index;

        if (localRoot) {
            candidates =
                localRoot.querySelectorAll(
                    selectors
                );

            for (
                index = 0;
                index < candidates.length;
                index += 1
            ) {
                if (
                    isNativeMobileNumberError(
                        candidates[index]
                    )
                ) {
                    return candidates[index];
                }
            }
        }

        candidates =
            document.querySelectorAll(
                selectors
            );

        for (
            index = 0;
            index < candidates.length;
            index += 1
        ) {
            if (
                isNativeMobileNumberError(
                    candidates[index]
                )
            ) {
                return candidates[index];
            }
        }

        return null;
    }

    function validateMobileNumber(
        showError
    ) {
        var field =
            findVisibleMobileNumberField();

        if (!field) {
            return true;
        }

        if (
            isValidUkMobileNumber(
                field.value
            )
        ) {
            hideNativeMobileNumberError(
                field
            );

            return true;
        }

        if (showError) {
            showNativeMobileNumberError(
                field
            );
        }

        return false;
    }

    function refreshMobileNumberValidation(
        field
    ) {
        window.setTimeout(
            function () {
                if (
                    field.dataset
                        .illuminexMobileValidationActive ===
                        "true"
                ) {
                    validateMobileNumber(
                        true
                    );
                }
            },
            0
        );
    }

        var field =
            findVisibleMobileNumberField();

        if (!field) {
            return;
        }

        field.scrollIntoView({
            behavior: "smooth",
            block: "center"
        });

    }

                    .illuminexMobileValidationActive =
                    "true";

                    field
                );
            },
            true
        );

        field.addEventListener(
            "blur",
            function () {
                field.dataset
                    .illuminexMobileValidationActive =
                    "true";

                refreshMobileNumberValidation(
                    field
                );
            },
            true
        );
    }

       CANDIDATE PRIVACY NOTICE LINK
       ====================================================== */

    function normaliseCandidatePrivacyText(
        value
    ) {
        return String(value || "")
            .replace(/\s+/g, " ")
            .trim()
            .toLowerCase();
    }

    function isCandidatePrivacyLink(
        link
    ) {
        if (
            !link ||
            link.tagName !== "A"
        ) {
            return false;
        }

        var label =
            normaliseCandidatePrivacyText(
                link.textContent
            );

        var href =
            normaliseCandidatePrivacyText(
                link.getAttribute("href")
            );

        return (
            link.getAttribute(
                "data-illuminex-candidate-privacy-link"
            ) === "true" ||
            label ===
                "candidate privacy notice" ||
            label ===
                "gdpr candidate agreement" ||
            href.indexOf(
                "candidate-privacy-notice.aspx"
            ) !== -1 ||
            href.indexOf(
                "gdpr-candidate-agreement"
            ) !== -1
        );
    }

    function updateCandidatePrivacyHeading() {
        if (
            !/candidate-privacy-notice\.aspx/i.test(
                window.location.pathname
            )
        ) {
            return;
        }

        document.querySelectorAll(
            "h1, h2, .page-title, .title"
        ).forEach(function (heading) {
            var headingText =
                normaliseCandidatePrivacyText(
                    heading.textContent
                );

            if (
                headingText ===
                    "gdpr candidate agreement"
            ) {
                heading.textContent =
                    "Candidate Privacy Notice";
            }
        });

        document.title =
            document.title.replace(
                /GDPR Candidate Agreement/gi,
                "Candidate Privacy Notice"
            );
    }

    function updateCandidatePrivacyLinks() {
        document.querySelectorAll(
            "a"
        ).forEach(function (link) {
            if (
                !isCandidatePrivacyLink(link)
            ) {
                return;
            }

            if (
                link.textContent.trim() !==
                    "Candidate Privacy Notice"
            ) {
                link.textContent =
                    "Candidate Privacy Notice";
            }

            if (
                link.getAttribute("href") !==
                    CANDIDATE_PRIVACY_URL
            ) {
                link.setAttribute(
                    "href",
                    CANDIDATE_PRIVACY_URL
                );
            }

            if (
                link.getAttribute("target") !==
                    "_blank"
            ) {
                link.setAttribute(
                    "target",
                    "_blank"
                );
            }

            if (
                link.getAttribute("rel") !==
                    "noopener noreferrer"
            ) {
                link.setAttribute(
                    "rel",
                    "noopener noreferrer"
                );
            }

            link.setAttribute(
                "data-illuminex-candidate-privacy-link",
                "true"
            );

            link.setAttribute(
                "title",
                "Read the Illuminex Consultancy Candidate Privacy Notice"
            );

            link.setAttribute(
                "aria-label",
                "Read the Illuminex Consultancy Candidate Privacy Notice in a new tab"
            );

            if (
                link.hasAttribute("onclick")
            ) {
                link.removeAttribute(
                    "onclick"
                );
            }
        });

        updateCandidatePrivacyHeading();
    }

    function interceptCandidatePrivacyLink(

    ) {

            event.target;

                "noopener,noreferrer"
            );

        }
    }

        }

            document.querySelectorAll(

            );

        var index;

        var label;

        for (

            index < controls.length;
            index += 1
        ) {

            if (
                label === "more vacancies" ||
                label === "view more vacancies" ||

            ) {
                return control;
            }

        ) {

                    (control.id || "") + " " +
                    (control.name || "") + " " +
                    (control.getAttribute("href") || "")
                )
            ) {

            }

        return null;
    }

    /* ======================================================
       LOGOUT REPAIR
       ====================================================== */

    function extractLogoutTarget(
        logoutLink
    ) {
        var href =
            logoutLink.getAttribute(
                "href"
            ) || "";

        var onclick =
            logoutLink.getAttribute(
                "onclick"
            ) || "";

        var source =
            href + " " + onclick;

        var match =
            source.match(
                /WebForm_PostBackOptions\(\s*['"]([^'"]+)['"]/i
            );

        if (
            match &&
            match[1]
        ) {
            return match[1];
        }

        var storedTarget =
            logoutLink.getAttribute(
                "data-illuminex-logout-target"
            );

        return (
            storedTarget ||
            DEFAULT_LOGOUT_TARGET
        );
    }

    function createSafeLogoutHref(
        logoutTarget
    ) {
        var safeTarget =
            logoutTarget.replace(
                /'/g,
                "\\'"
            );

        return (
            "javascript:" +
            "WebForm_DoPostBackWithOptions(" +
            "new WebForm_PostBackOptions(" +
            "'" + safeTarget + "'," +
            "''," +
            "false," +
            "''," +
            "''," +
            "false," +
            "true" +
            ")" +
            ");"
        );
    }

    function repairLogoutLink(
        logoutLink
    ) {
        if (!logoutLink) {
            return;
        }

        var logoutTarget =
            extractLogoutTarget(
                logoutLink
            );

        var safeHref =
            createSafeLogoutHref(
                logoutTarget
            );

        logoutLink.setAttribute(
            "data-illuminex-logout-target",
            logoutTarget
        );

        logoutLink.setAttribute(
            "data-illuminex-logout-fixed",
            "true"
        );

        if (
            logoutLink.getAttribute(
                "href"
            ) !== safeHref
        ) {
            logoutLink.setAttribute(
                "href",
                safeHref
            );
        }

        if (
            logoutLink.hasAttribute(
                "onclick"
            )
        ) {
            logoutLink.removeAttribute(
                "onclick"
            );
        }
    }

    function repairAllLogoutLinks() {
        document.querySelectorAll(
            LOGOUT_SELECTOR
        ).forEach(function (logoutLink) {
            repairLogoutLink(
                logoutLink
            );
        });
    }

    function findLogoutLink(event) {
        var target =
            event.target;

        if (!(target instanceof Element)) {
            return null;
        }

        return target.closest(
            LOGOUT_SELECTOR
        );
    }

    function prepareLogout(event) {
        var logoutLink =
            findLogoutLink(event);

        if (!logoutLink) {
            return;
        }

        repairLogoutLink(
            logoutLink
        );
    }

    function performSafeLogout(event) {
        var logoutLink =
            findLogoutLink(event);

        if (!logoutLink) {
            return;
        }

        event.preventDefault();
        event.stopPropagation();
        event.stopImmediatePropagation();

        var logoutTarget =
            extractLogoutTarget(
                logoutLink
            );

        repairLogoutLink(
            logoutLink
        );

        if (
            typeof window
                .WebForm_DoPostBackWithOptions ===
                "function" &&
            typeof window
                .WebForm_PostBackOptions ===
                "function"
        ) {
            window.WebForm_DoPostBackWithOptions(
                new window.WebForm_PostBackOptions(
                    logoutTarget,
                    "",
                    false,
                    "",
                    "",
                    false,
                    true
                )
            );

            return;
        }

        if (
            typeof window.__doPostBack ===
            "function"
        ) {
            window.__doPostBack(
                logoutTarget,
                ""
            );
        }
    }

    function initialise() {
        installCvUploadModalStyles();
        updateCvUploadModalState();
        installGuidanceStyles();
        updateApplicationLoginMessage();
        installApplicationErrorWeightStyles();
        styleLoginFailureMessage();
        addNewApplicantPasswordGuidance();

        bindMobileNumberValidation();
        updateCandidatePrivacyLinks();
        hideDuplicateHeaderAuthLinks();
        updateCancelButton();
        updateOpportunitiesButton();
        updateSuccessMoreVacanciesControl();
        removeUnwantedSocials();
        repairAllLogoutLinks();
    }

    function scheduleInitialise() {
        if (initialiseScheduled) {
            return;
        }

        initialiseScheduled = true;

        window.requestAnimationFrame(
            function () {
                initialiseScheduled =
                    false;

                initialise();
            }
        );
    }

    function start() {
        initialise();

        document.addEventListener(
            "pointerdown",
            prepareLogout,
            true
        );

        document.addEventListener(
            "touchstart",
            prepareLogout,
            true
        );

        document.addEventListener(
            "click",
            interceptCandidatePrivacyLink,
            true
        );

        document.addEventListener(
            "click",
            interceptInvalidMobileNumberClick,
            true
        );

        document.addEventListener(
            "submit",
            interceptInvalidMobileNumberSubmit,
            true
        );

        document.addEventListener(
            "keydown",
            submitCvUploadWithEnter,
            true
        );

        document.addEventListener(
            "click",
            performSafeLogout,
            true
        );

        document.addEventListener(
            "click",
            interceptSuccessMoreVacancies,
            true
        );

        document.addEventListener(
            "keydown",
            function (event) {
                if (
                    event.key !== "Enter" &&
                    event.key !== " "
                ) {
                    return;
                }

                var logoutLink =
                    findLogoutLink(event);

                if (!logoutLink) {
                    return;
                }

                performSafeLogout(
                    event
                );
            },
            true
        );

        var observer =
            new MutationObserver(
                function () {
                    scheduleInitialise();
                }
            );

        observer.observe(
            document.documentElement,
            {
                childList: true,
                subtree: true,
                attributes: true,
                attributeFilter: [
                    "href",
                    "onclick",
                    "style",
                    "class"
                ]
            }
        );

        window.addEventListener(
            "load",
            scheduleInitialise
        );

        window.addEventListener(
            "pageshow",
            scheduleInitialise
        );

        setTimeout(
            scheduleInitialise,
            250
        );

        setTimeout(
            scheduleInitialise,
            750
        );

        setTimeout(
            scheduleInitialise,
            1500
        );

        setTimeout(
            scheduleInitialise,
            3000
        );
    }

    if (
        document.readyState ===
        "loading"
    ) {
        document.addEventListener(
            "DOMContentLoaded",
            start
        );
    } else {
        start();
    }
})();
